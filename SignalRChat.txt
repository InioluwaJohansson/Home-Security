private MessageDto MapToDto(Message message)
{
    return new MessageDto
    {
        Id = message.Id,
        ChatId = message.ChatId,
        SenderPersonId = message.SenderPersonId,
        ReplyMessageId = message.ReplyToMessageId,
        Content = message.Content,
        Type = message.Type,
        IsEdited = message.IsEdited,
        IsDeleted = message.IsDeleted,
        SentAt = message.SentAt,
        SenderName = $"{message.SenderPerson.PersonDetails.FirstName} {message.SenderPerson.PersonDetails.LastName}",
        SenderProfileImage = ConvertImageToBase64(message.SenderPerson.PersonDetails.ImageUrl) ?? "",
        Attachments = message.Attachments?
            .Select(a => new GetMessageAttachmentDto
            {
                FileName = a.FileName,
                FilePath = a.FilePath,
                ContentType = a.ContentType,
                FileSize = a.FileSize,
                ThumbnailPath = a.ThumbnailPath
            })
            .ToList() ?? new List<GetMessageAttachmentDto>()
    };
}

private ChatDto MapToDtoChat(Chat chat)
{
    return new ChatDto
    {
        Id = chat.Id,
        Name = chat.Name,
        Description = chat.Description,
        IsGroup = chat.IsGroup,
        ImageUrl = ConvertImageToBase64(chat.ImageUrl),
        CreatedAt = chat.CreatedAt,
        Participants = chat.Participants?
            .Select(p => new ChatParticipantDto
            {
                PersonId = p.PersonId,
                IsAdmin = p.IsAdmin,
                IsOnline = false,
                FullName = "",
                ProfileImageUrl = null
            })
            .ToList() ?? new List<ChatParticipantDto>(),
        LastMessage = chat.Messages?
            .OrderByDescending(x => x.SentAt)
            .Select(x => new MessageDto
            {
                Id = x.Id,
                ChatId = x.ChatId,
                SenderPersonId = x.SenderPersonId,
                Content = x.Content,
                Type = x.Type,
                IsEdited = x.IsEdited,
                IsDeleted = x.IsDeleted,
                SentAt = x.SentAt,
                SenderName = "",
                SenderProfileImage = null,
                Attachments = x.Attachments?
                .Select(a => new GetMessageAttachmentDto
                {
                    FileName = a.FileName,
                    FilePath = a.FilePath,
                    ContentType = a.ContentType,
                    FileSize = a.FileSize,
                    ThumbnailPath = a.ThumbnailPath
                })
                .ToList() ?? new List<GetMessageAttachmentDto>()
            }).FirstOrDefault(),
        UnreadCount = 0
    };
}
